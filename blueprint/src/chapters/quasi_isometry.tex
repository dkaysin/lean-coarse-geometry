\section{Maps at finite distance}

\inputleannode{def:finite-dist}
\inputleannode{lem:finite-dist-refl}
\inputleannode{lem:finite-dist-symm}
\inputleannode{lem:finite-dist-trans}

\begin{lemma}[L\"oh, Prop.~5.1.11 (3a)]
  \label{lem:finite-dist-comp-right}
  \uses{def:finite-dist}
  If $f$ and $g \colon X \to Y$ have finite distance, then so do $f \circ k$ and $g \circ k$ for
  every map $k \colon W \to X$.
  \plannedlean{Metric.HasFiniteDist.comp_right}
\end{lemma}
\begin{proof}
  The same constant works.
\end{proof}

\begin{lemma}[L\"oh, Prop.~5.1.11 (3b)]
  \label{lem:finite-dist-comp-left}
  \uses{def:finite-dist, def:coarse-lipschitz}
  If $f$ and $g \colon X \to Y$ have finite distance $D$ and $h \colon Y \to Z$ is $(L, C)$-coarse
  Lipschitz, then $h \circ f$ and $h \circ g$ have finite distance $L D + C$.
  \plannedlean{Metric.HasFiniteDist.comp_left}
\end{lemma}
\begin{proof}
  $d(h f x, h g x) \le L\,d(f x, g x) + C$.
\end{proof}

\begin{lemma}[L\"oh, Prop.~5.1.11 (1)]
  \label{lem:coarse-lipschitz-of-finite-dist}
  \uses{def:finite-dist, def:coarse-lipschitz, def:coarse-antilipschitz}
  If $f$ is $(L, C)$-coarse Lipschitz and $g$ has finite distance $D$ from $f$, then $g$ is
  $(L, C + 2D)$-coarse Lipschitz. If $f$ is $(L, C)$-coarse antilipschitz, then $g$ is
  $(L, C + 2LD)$-coarse antilipschitz.
  \plannedlean{Metric.CoarseLipschitzWith.of_hasFiniteDist}
  \plannedlean{Metric.CoarseAntilipschitzWith.of_hasFiniteDist}
\end{lemma}
\begin{proof}
  Triangle inequality: $d(g x, g y)$ and $d(f x, f y)$ differ by at most $2D$.
\end{proof}

\begin{lemma}
  \label{lem:isometry-qie}

  An isometric embedding is a $(1, 0)$-quasi-isometric embedding.
  \plannedlean{Isometry.isQuasiIsometricEmbeddingWith}

  Both bounds hold with equality.

\begin{lemma}[L\"oh, Prop.~5.1.11 (4)]
  \label{lem:qie-comp}
  \uses{def:qie, lem:qie-of-bounds, lem:coarse-lipschitz-comp, lem:coarse-antilipschitz-comp}

  \plannedlean{Metric.IsQuasiIsometricEmbedding.comp}

\begin{proof}
  Compose the two bounds separately and merge the constants with
  Lemma~\ref{lem:qie-of-bounds}.

\section{Quasi-isometries}

\inputleannode{def:qi}
\inputleannode{def:quasi-isometric}

\begin{lemma}[L\"oh, Prop.~5.1.11 (5)]
  \label{lem:qi-id}
  \uses{def:qi, lem:isometry-qie, lem:finite-dist-refl}
  The identity map, and more generally every isometric bijection, is a quasi-isometry.
  \plannedlean{IsometryEquiv.isQuasiIsometry}
\end{lemma}
\begin{proof}
  The inverse is a quasi-inverse, with both composites equal to the identity.
\end{proof}

\begin{lemma}[L\"oh, Ex.~5.1.7]
  \label{lem:qi-comp}
  \uses{def:qi, lem:qie-comp, lem:finite-dist-comp-right, lem:finite-dist-comp-left,
    lem:finite-dist-trans}
  The composition of two quasi-isometries is a quasi-isometry.
  \plannedlean{Metric.IsQuasiIsometry.comp}
\end{lemma}
\begin{proof}
  If $g'$ and $f'$ are quasi-inverses of $g$ and $f$, then $f' \circ g'$ is a quasi-inverse of
  $g \circ f$: $f' g' g f$ is at finite distance from $f' f$ (precompose and postcompose), hence
  from the identity.
\end{proof}

\begin{lemma}[L\"oh, Prop.~5.1.11 (2)]
  \label{lem:qi-of-finite-dist}
  \uses{def:qi, lem:coarse-lipschitz-of-finite-dist, lem:qie-of-bounds, lem:finite-dist-trans,
    lem:finite-dist-comp-left}
  A map at finite distance from a quasi-isometry is a quasi-isometry.
  \plannedlean{Metric.IsQuasiIsometry.of_hasFiniteDist}
\end{lemma}
\begin{proof}
  The quasi-inverse of the given quasi-isometry also works for the new map.
\end{proof}

\begin{theorem}
  \label{thm:quasi-isometric-equivalence}
  \uses{def:quasi-isometric, lem:qi-id, lem:qi-comp}
  Being quasi-isometric is an equivalence relation on pseudometric spaces.
  \plannedlean{Metric.QuasiIsometric.refl, .symm, .trans}
\end{theorem}
\begin{proof}
  Reflexivity is Lemma~\ref{lem:qi-id}, symmetry is immediate from the definition (the
  quasi-inverse is a quasi-isometry), transitivity is Lemma~\ref{lem:qi-comp}.
\end{proof}
