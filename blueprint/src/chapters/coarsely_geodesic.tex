\chapter{Coarsely geodesic spaces}
\label{chap:coarsely-geodesic}

Milestone M2. An \emph{$r$-chain} from $x$ to $y$ is a sequence $x = x_0, \dots, x_n = y$ with
$d(x_i, x_{i+1}) \le r$; $n$ is its number of steps. The encoding of chains (a map
$\mathrm{Fin}(n + 1) \to X$ or the reflexive--transitive closure of a relation) and the split
into two definitions are open decisions (design decisions D5, D6, D13), so the definitions are not
ready.

\begin{definition}[Coarsely connected space]
  \label{Metric.IsCoarselyConnectedWith}
  \notready
  A pseudometric space $X$ is \emph{$r$-coarsely connected} if any two points are joined by an
  $r$-chain.
\end{definition}

\begin{definition}[Coarsely geodesic space]
  \label{Metric.IsCoarselyGeodesicWith}
  \notready
  A pseudometric space $X$ is \emph{$(r, L, C)$-coarsely geodesic} if any two points $x, y$ are
  joined by an $r$-chain with at most $L\,d(x, y) + C$ steps.
\end{definition}

\begin{lemma}
  \label{NormedSpace.isCoarselyGeodesicWith}
  \notready
  \uses{Metric.IsCoarselyGeodesicWith}
  A real normed space is $(r, 1/r, 1)$-coarsely geodesic for every $r > 0$.
\end{lemma}
\begin{proof}
  Take the points $x + \frac{k}{n}(y - x)$ for $k = 0, \dots, n$ and
  $n = \lceil \|y - x\| / r \rceil$.
\end{proof}

\begin{lemma}
  \label{Group.Generators.isCoarselyGeodesicWith}
  \notready
  \uses{Metric.IsCoarselyGeodesicWith, Group.Generators.groupNorm}
  A group with the word metric of a generating family is $(1, 1, 0)$-coarsely geodesic.
\end{lemma}
\begin{proof}
  The prefixes of a geodesic word for $g^{-1} h$, multiplied on the left by $g$, form a
  $1$-chain from $g$ to $h$ with $|g^{-1} h|$ steps.
\end{proof}
