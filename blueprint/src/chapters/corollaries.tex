\begin{corollary}[Change of generators]
  \label{cor:change-of-generators}
  \notready
  \uses{lem:change-of-generators-bound, def:word-metric, def:qi, lem:qie-of-bounds}
  For finite generating families $P$ and $Q$ of $G$, the identity $(G, d_P) \to (G, d_Q)$ is
  bilipschitz, hence a quasi-isometry.
\end{corollary}
\begin{proof}
  Lemma~\ref{lem:change-of-generators-bound} in both directions.
\end{proof}

\begin{corollary}[Finite-index subgroups]
  \label{cor:finite-index-qi}
  \notready
  \uses{thm:svarc-milnor-qi, lem:word-metric-coarsely-geodesic, lem:word-metric-finite-balls,
    lem:finite-group-metrically-proper, lem:finite-index-cobounded}
  If $G$ is finitely generated and $H \le G$ has finite index, then the inclusion $H \to G$ is a
  quasi-isometry for any finite generating families of $H$ and $G$.
\end{corollary}
\begin{proof}
  $H$ acts on $(G, d_P)$ by left multiplication; the action is metrically proper by finiteness
  of balls and cobounded by finite index. Apply Theorem~\ref{thm:svarc-milnor-qi} with
  $x_0 = 1$.
\end{proof}

\begin{corollary}[$\Z^n$ and $\R^n$, L\"oh Ex.~5.2.8]
  \label{cor:zn-qi-rn}
  \notready
  \uses{thm:svarc-milnor-qi, lem:normed-space-coarsely-geodesic,
    lem:properly-discontinuous-metrically-proper}
  $\Z^n$ with the word metric of the standard basis is quasi-isometric to $\R^n$.
\end{corollary}
\begin{proof}
  $\Z^n$ acts on $\R^n$ (sup norm) by translations, properly discontinuously and with the unit
  cube as a bounded fundamental set.
\end{proof}
