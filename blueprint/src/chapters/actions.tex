\begin{lemma}[Cocompact actions are cobounded]
  \label{isCoboundedSMul_of_compactSpace_quotient}
  \uses{IsCoboundedSMul}
  If $G$ acts isometrically on a pseudometric space $X$ and the orbit space $X/G$, with the
  quotient topology, is compact, then the action is cobounded.
\end{lemma}
\begin{proof}
  If $X$ is empty, the empty set is bounded and covers $X$. Otherwise fix $x_0 \in X$. The
  distance to the orbit $G x_0$ is a continuous function on $X$, constant on orbits, so it
  induces a continuous function on the compact space $X/G$, which is bounded, say by $\rho$. So
  every point is within $\rho + 1$ of a point $g x_0$, that is, lies in $g\,\bar B(x_0, \rho + 1)$.
  (This is \texttt{exists\_smul\_closedBall\_eq\_univ} in the copied coarse-space library.)
\end{proof}
