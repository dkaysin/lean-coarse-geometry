\chapter{Hypotheses on the action}
\label{chap:actions}

Milestone M4. $G$ acts on a space $X$ with a bornology, for instance the bounded sets of a
pseudometric space; the definitions use only bounded sets (design decision D14).

\begin{definition}[Cobounded action]
  \label{IsCoboundedSMul}
  \notready
  The action is \emph{cobounded} if some bounded set $s \subseteq X$ satisfies
  $\bigcup_{g \in G} g s = X$.
\end{definition}

\inputleannode{IsMetricallyProperSMul}

\begin{lemma}[Orbit map]
  \label{isMetricallyProperSMul_iff_tendsto}
  \notready
  \uses{IsMetricallyProperSMul}
  For an isometric action and any $x_0 \in X$, the action is metrically proper if and only if
  the orbit map $g \mapsto g x_0$ sends the cofinite filter to the cobounded filter.
\end{lemma}
\begin{proof}
  ($\Rightarrow$) $\{g \mid d(x_0, g x_0) \le R\}$ is contained in $\{g \mid g B \cap B \ne
  \emptyset\}$ for $B$ the closed ball of radius $R$ about $x_0$. ($\Leftarrow$) If
  $s \subseteq \bar B(x_0, R)$ and $g s \cap s \ne \emptyset$, then $d(x_0, g x_0) \le 2R$.
\end{proof}

\begin{lemma}
  \label{isMetricallyProperSMul_of_properlyDiscontinuousSMul}
  \notready
  \uses{IsMetricallyProperSMul}
  A properly discontinuous action on a proper metric space is metrically proper.
\end{lemma}
\begin{proof}
  A bounded set lies in a closed ball, which is compact; apply the finiteness in the definition
  of a properly discontinuous action.
\end{proof}

\begin{lemma}
  \label{isMetricallyProperSMul_of_finite}
  \notready
  \uses{IsMetricallyProperSMul}
  Every action of a finite group is metrically proper, and the restriction of a metrically
  proper action to a subgroup is metrically proper.
\end{lemma}
\begin{proof}
  Subsets of finite sets are finite.
\end{proof}

\begin{lemma}
  \label{IsCoboundedSMul.of_finiteIndex}
  \notready
  \uses{IsCoboundedSMul}
  If $G$ acts coboundedly and $H \le G$ has finite index, then $H$ acts coboundedly.
\end{lemma}
\begin{proof}
  If $s$ works for $G$ and $g_1, \dots, g_k$ are coset representatives of $G / H$ (on the
  appropriate side), then $\bigcup_j g_j s$ is bounded and works for $H$.
\end{proof}
